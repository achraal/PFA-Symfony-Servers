Suite de l'application symfony :

nano src/Controller/Controller.php

<?php
namespace App\Controller; 
use Symfony\Component\HttpFoundation\Response;
use Symfony\Bundle\FrameworkBundle\Controller\AbstractController;
use App\Repository\ProduitRepository;
use Symfony\Component\HttpFoundation\Request;
use Doctrine\ORM\EntityManagerInterface;
use App\Entity\Produit;

class Controller extends AbstractController
{
    public function number(): Response
    {
        $number = random_int(0, 100);
        return new Response(
            '<html><body>Lucky number: '.$number.'</html></body>'
        );
    }

    public function home(ProduitRepository $produitRepository)
    {
        $produits = $produitRepository->findAll();
        return $this->render("home.html.twig",["produits"=>$produits]);
    }

    public function ajout()
    {
        return $this->render("ajout.html.twig");
    }

    public function ajoutProduit(Request $request, EntityManagerInterface $entityManager)
    {
        $nom= $request->request->get("nom");
        $description= $request->request->get("description");
        $qte= $request->request->get("qte");

        $produit = new Produit();

        $produit->setNom($nom);
        $produit->setDescription($description);
        $produit->setQte($qte);

        $entityManager->persist($produit);
        $entityManager->flush();

        return $this->render("ajoutProduit.html.twig");

    }

    public function modifier(int $id, EntityManagerInterface $entityManager,Request $request)
    {    
         $produit=$entityManager->getRepository(Produit ::class)->find($id);
         return $this->render('modifier.html.twig', ['produit' => $produit]);
    }

    public function modifierProduit(EntityManagerInterface $entityManager, Request $request)
    {
        $id = $request->request->get("id");
        $produit = $entityManager->getRepository(Produit::class)->find($id);
        $nom = $request->request->get("nom");
        $description = $request->request->get("description");
        $qte = $request->request->get("quantite");

        $produit->setNom($nom);
        $produit->setDescription($description);
        $produit->setQte((int)$qte);

        $entityManager->flush();
        return $this->redirect('/home');
    }

    public function supprimer(int $id, EntityManagerInterface $entityManager, Request $request)
    {
        $produit = $entityManager->getRepository(Produit::class)->find($id);
        $entityManager->remove($produit);
        $entityManager->flush();
        return $this->redirect('/home');
    }
}

Configuration d'une adresse ip statique

nmcli connection add type ethernet ifname enp4s0f0 con-name enp4s0f0
nmcli connection show      # Afficher les connexions existantes

# Créer une nouvelle connexion ou modifier une existante
nmcli connection modify enp4s0f0 ipv4.addresses 192.168.100.5/24
nmcli connection modify enp4s0f0 ipv4.gateway 192.168.100.1
nmcli connection modify enp4s0f0 ipv4.dns 192.168.100.2
nmcli connection modify enp4s0f0 ipv4.method manual

# Activer la connexion
nmcli connection up enp4s0f0
nmcli device show enp4s0f0
ip address show enp4s0f0

*fermez le fichier ensuite

nano /var/named/emsi.ma.zone
$TTL 1D
@	IN SOA	achraf.emsi.ma. root.emsi.ma. (   # remplacer achraf par votre hostname 
					0	; serial
					1D	; refresh
					1H	; retry
					1W	; expire
					3H )	; minimum
		IN	NS	achraf.emsi.ma.  # remplacer achraf par votre hostname

Exemple de configuration du DHCP :

cp /usr/share/doc/dhcp-server/dhcpd.conf.example /etc/dhcp/dhcp.conf
nano /etc/dhcp/dhcp.conf

subnet 192.168.100.0 netmask 255.255.255.0 {  
  range 192.168.100.20 192.168.100.50;
  option domain-name-servers 192.168.100.5;
  option domain-name "emsi.ma";
  option routers 192.168.100.1;
  option broadcast-address 192.168.100.255;
  option subnet-mask 255.255.255.0;
  default-lease-time 600;
  max-lease-time 7200;
}

systemctl restart dhcpd
systemctl enable dhcpd

Configuration du Phpmyadmin

Alias /phpmyadmin /usr/share/phpMyAdmin
<Directory /usr/share/phpMyAdmin/>
    AddDefaultCharset UTF-8
    <IfModule mod_authz_core.c>
        # Apache 2.4
        Require all granted
    </IfModule>
    <IfModule !mod_authz_core.c>
        # Apache 2.2
        Order Deny,Allow
        Deny from All
        Allow from 127.0.0.1
        Allow from ::1
    </IfModule>
</Directory>
